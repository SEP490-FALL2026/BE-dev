SELECT s."workspace_id", s."plan_id", count(*) AS "periods"
  FROM "subscriptions" s
  JOIN "plans" p ON p."id" = s."plan_id"

   AND NOT EXISTS (SELECT 1 FROM "subscriptions" n WHERE n."renewed_from_id" = s."id")
